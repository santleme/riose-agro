########################################Mod Evaluation 
RMSE <- function(o,p){
  o <- Database$CH4_MJ_Fin
  p <- Database$CH4_Pred_M
  meano <- mean(o, na.rm=TRUE)
  meanp <- mean(p, na.rm=TRUE)
  res <- o - p
  res2 <- res^2
  rm <- sqrt(mean(res2, na.rm = TRUE))
  uss <- sum(res2, na.rm = TRUE)
  lo <- ifelse(is.na(o) == FALSE, 1, 0)
  n <- sum(lo)
  mb <- sum(res, na.rm = TRUE) / n
  sse <- anova(lm(res ~ p))[2, 2]
  msb <- mb^2
  mspe <- rm^2
  msre <- sse / n
  msslope <- mspe - msre - msb
  mean_val <- msb / mspe * 100
  slope <- msslope / mspe * 100
  residual <- msre / mspe * 100
  rsr <- rm / sd(o, na.rm = TRUE)
  ccc <- epi.ccc(o, p)$rho.c[1]
  rmp <- rm / meano * 100
  output <- format(c(meano, meanp, rmp, rm, mean_val, slope, residual, mb, msslope, ccc))
  labels <- c("Observed Mean", "Predicted Mean", "RMSE, % Mean", "RMSE, MJ/d", "Mean Bias, % MSE", "Slope Bias, % MSE", "Residual Error, % MSE", "Mean Bias", "Slope Bias", "CCC")
  out <- data.frame(labels, output)
  return(out)
}
